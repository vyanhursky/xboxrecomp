cmake_minimum_required(VERSION 3.20)
project(usb_transfer_regression C)
if(NOT WIN32)
    message(FATAL_ERROR "Synthetic memory fixture currently uses Windows VirtualAlloc")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(usb-transfer-test test_main.c)
target_link_libraries(usb-transfer-test PRIVATE xboxrecomp)
if(XBOXRECOMP_SDL3)
    target_compile_definitions(usb-transfer-test PRIVATE XBOXRECOMP_HAVE_SDL3=1)
    target_link_libraries(usb-transfer-test PRIVATE SDL3::SDL3-static)
endif()
target_compile_definitions(usb-transfer-test PRIVATE _CRT_SECURE_NO_WARNINGS)
enable_testing()
add_test(NAME usb-transfers COMMAND usb-transfer-test)
add_test(NAME usb-script-axes COMMAND usb-transfer-test --script-axes)
if(XBOXRECOMP_SDL3)
    add_test(NAME usb-script-axes-virtual COMMAND usb-transfer-test --script-axes-virtual)
endif()
