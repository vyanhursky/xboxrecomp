cmake_minimum_required(VERSION 3.20)
project(nv2a_raw_draw C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_executable(nv2a-raw-plan-test plan_test.c)
target_include_directories(nv2a-raw-plan-test PRIVATE ../../src/kernel)
add_test(NAME nv2a-raw-plan COMMAND nv2a-raw-plan-test)
add_executable(nv2a-raw-exec-test exec_test.c
    ../../src/kernel/nv2a_pb_exec.c ../../src/kernel/nv2a_vsh_cpu.c
    ../../src/kernel/nv2a_vsh_interp.c ../../src/kernel/nv2a_combiner.c
    ../../src/kernel/nv2a_flip_hold.c ../nv2a_exec_stubs/fork_stubs.c)
target_include_directories(nv2a-raw-exec-test PRIVATE ../../src/kernel ../../src)
target_compile_definitions(nv2a-raw-exec-test PRIVATE NV2A_RAW_DRAW_TEST)
if(MSVC)
    target_compile_definitions(nv2a-raw-exec-test PRIVATE _CRT_SECURE_NO_WARNINGS)
elseif(NOT WIN32)
    add_subdirectory(../../src/platform platform)
    target_link_libraries(nv2a-raw-exec-test PRIVATE platform m pthread)
endif()
add_test(NAME nv2a-raw-exec-off COMMAND nv2a-raw-exec-test 0)
add_test(NAME nv2a-raw-exec-native COMMAND nv2a-raw-exec-test 1)
add_test(NAME nv2a-raw-exec-fallback COMMAND nv2a-raw-exec-test 2)
if(WIN32)
    add_executable(nv2a-raw-adapter-test adapter_test.c
        ../../src/kernel/nv2a_vsh_cpu.c ../../src/kernel/nv2a_vsh_interp.c)
    target_include_directories(nv2a-raw-adapter-test PRIVATE ../../src/kernel ../../src/d3d)
    target_compile_definitions(nv2a-raw-adapter-test PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(nv2a-raw-adapter-test PRIVATE d3d11 d3dcompiler)
    add_test(NAME nv2a-raw-adapter-warp COMMAND nv2a-raw-adapter-test)
endif()
option(XBOXRECOMP_RAW_VULKAN_TEST "Build the headless native Vulkan bridge test (needs Vulkan and shaderc)" OFF)
if(NOT WIN32 AND XBOXRECOMP_RAW_VULKAN_TEST)
    find_package(Vulkan REQUIRED)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_path(RAW_SHADERC_HEADERS shaderc/shaderc.h)
    find_library(RAW_SHADERC_LIBRARY NAMES shaderc_shared shaderc)
    if(NOT RAW_SHADERC_HEADERS OR NOT RAW_SHADERC_LIBRARY)
        message(FATAL_ERROR "The headless Vulkan fixture needs shaderc headers/library (for example libshaderc-dev)")
    endif()
    add_subdirectory(../../src/d3d raw-d3d)
    get_target_property(raw_defs xbox_d3d8 INTERFACE_COMPILE_DEFINITIONS)
    if(NOT "XBOXRECOMP_HAVE_VULKAN=1" IN_LIST raw_defs)
        message(FATAL_ERROR "The raw draw Vulkan fixture requires the Vulkan/shaderc backend")
    endif()
    add_executable(nv2a-raw-vulkan-test vulkan_test.c
        ../../src/kernel/nv2a_vsh_cpu.c ../../src/kernel/nv2a_vsh_interp.c)
    target_include_directories(nv2a-raw-vulkan-test PRIVATE ../../src/kernel ../../src/d3d)
    target_link_libraries(nv2a-raw-vulkan-test PRIVATE xbox_d3d8 platform m)
    add_test(NAME nv2a-raw-validation-runner COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/validation_run_test.py)
    add_test(NAME nv2a-raw-vulkan COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/validation_run.py $<TARGET_FILE:nv2a-raw-vulkan-test>)
    add_test(NAME nv2a-raw-vulkan-msaa COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/validation_run.py $<TARGET_FILE:nv2a-raw-vulkan-test> --msaa)
endif()
