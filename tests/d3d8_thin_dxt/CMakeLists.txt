cmake_minimum_required(VERSION 3.20)
project(d3d8_thin_dxt C)
if(NOT WIN32)
    message(FATAL_ERROR "This fixture exercises native D3D11 resource creation on Windows")
endif()
enable_testing()
add_executable(d3d8_thin_dxt_test test_main.c
    ../../src/d3d/d3d8_resources.c ../../src/d3d/d3d8_formats.c)
target_include_directories(d3d8_thin_dxt_test PRIVATE ../../src/d3d ../../src)
target_link_libraries(d3d8_thin_dxt_test PRIVATE d3d11 dxgi dxguid)
if(MSVC)
    target_compile_options(d3d8_thin_dxt_test PRIVATE /std:c11)
    target_compile_definitions(d3d8_thin_dxt_test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME d3d8_thin_dxt COMMAND d3d8_thin_dxt_test)
