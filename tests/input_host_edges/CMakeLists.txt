cmake_minimum_required(VERSION 3.20)
project(input_host_edges C)
set(CMAKE_C_STANDARD 11)
if(NOT WIN32)
    message(FATAL_ERROR "Windows native integration fixture")
endif()
list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/../../cmake)
if(SDL3_DIR)
    find_package(SDL3 REQUIRED CONFIG)
else()
    include(xbox_sdl3)
endif()
add_executable(input_host_edges test_main.c ../../src/input/input_host.c
    ../../src/input/input_edges.c ../../src/input/input_map.c ../../src/platform/host_time.c)
target_include_directories(input_host_edges PRIVATE ../../src/input ../../src)
target_compile_definitions(input_host_edges PRIVATE XBOXRECOMP_HAVE_SDL3=1)
target_link_libraries(input_host_edges PRIVATE SDL3::SDL3-static xinput winmm)
enable_testing()
add_test(NAME input_host_edges COMMAND input_host_edges)
